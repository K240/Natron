cmake_minimum_required(VERSION 3.26)
project(NatronHdNvpHost LANGUAGES CXX)
find_package(pxr CONFIG REQUIRED)
if(NOT PXR_VERSION EQUAL 2605)
    message(FATAL_ERROR "The hdNVP host requires its standalone OpenUSD 26.05 SDK")
endif()
add_executable(NatronHdNvpHost main.cpp)
target_compile_features(NatronHdNvpHost PRIVATE cxx_std_20)
target_compile_definitions(NatronHdNvpHost PRIVATE NDEBUG NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(NatronHdNvpHost PRIVATE /utf-8 /Zc:__cplusplus /Z7)
target_link_options(NatronHdNvpHost PRIVATE /DEBUG)
target_link_libraries(NatronHdNvpHost PRIVATE usd usdGeom usdImaging hd hdsi hf hgi cameraUtil js plug dbghelp)
set_target_properties(NatronHdNvpHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
